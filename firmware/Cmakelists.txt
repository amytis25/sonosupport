cmake_minimum_required(VERSION 3.13)

include(pico_sdk_import.cmake)

project(sonosupport C CXX ASM)
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 17)

pico_sdk_init()

add_executable(sonosupport
    main.c
    HAL/src/emg.c
)
target_include_directories(sonosupport PRIVATE
    ${CMAKE_CURRENT_SOURCE_DIR}/HAL/include
)

target_link_libraries(sonosupport
    pico_stdlib
    hardware_adc
)

pico_enable_stdio_usb(sonosupport 1)
pico_enable_stdio_uart(sonosupport 0)

pico_add_extra_outputs(sonosupport)
